33bool PostMachine::step() {
34 if (halted_) return true;
35 if (state_ >= program_.size()) { halted_ = true; return true; }
36
37 const PostInstruction instruction = program_[state_];
38 switch (instruction.op) {
39 case PostOp::Write0:
40 tape_[head_] = false;
41 ++state_;
42 break;
43 case PostOp::Write1:
44 tape_[head_] = true;
45 ++state_;
46 break;
47 case PostOp::MoveLeft:
48 if (head_ == std::numeric_limits<std::int64_t>::min())
49 throw std::overflow_error("Tape head position underflow.");
50 --head_;
51 ++state_;
52 break;
53 case PostOp::MoveRight:
54 if (head_ == std::numeric_limits<std::int64_t>::max())
55 throw std::overflow_error("Tape head position overflow.");
56 ++head_;
57 ++state_;
58 break;
59 case PostOp::JumpIfZero:
60 state_ = tapeAt(head_) ? state_ + 1 : instruction.target;
61 break;
62 case PostOp::JumpIfOne:
63 state_ = tapeAt(head_) ? instruction.target : state_ + 1;
64 break;
65 case PostOp::Halt:
66 halted_ = true;
67 break;
68 }
69 if (state_ >= program_.size() && !halted_) halted_ = true;
70 return halted_;
71}

139std::istream& operator>>(std::istream& in, PostMachine& m) {
140 std::string tag, op;
141 int version = 0, halted = 0, bit = 0;
142 std::int64_t head = 0, position = 0;
143 std::size_t state = 0, tapeCount = 0, programCount = 0;
144 if (!(in >> tag >> version) || tag != "POSTMACHINE" || version != 1) {
145 in.setstate(std::ios::failbit); return in;
146 }
147 if (!(in >> tag >> head) || tag != "HEAD" ||
148 !(in >> tag >> state) || tag != "STATE" ||
149 !(in >> tag >> halted) || tag != "HALTED" || (halted != 0 && halted != 1) ||
150 !(in >> tag >> tapeCount) || tag != "TAPE") {
151 in.setstate(std::ios::failbit); return in;
152 }
153 std::map<std::int64_t, bool> tape;
154 for (std::size_t i = 0; i < tapeCount; ++i) {
155 if (!(in >> position >> bit) || (bit != 0 && bit != 1)) {
156 in.setstate(std::ios::failbit); return in;
157 }
158 if (bit == 1) tape[position] = true;
159 }
160 if (!(in >> tag >> programCount) || tag != "PROGRAM") {
161 in.setstate(std::ios::failbit); return in;
162 }
163 std::vector<PostInstruction> program;
164 for (std::size_t i = 0; i < programCount; ++i) {
165 if (!(in >> op)) { in.setstate(std::ios::failbit); return in; }
166 if (op == "WRITE0") program.push_back(PostInstruction::write0());
167 else if (op == "WRITE1") program.push_back(PostInstruction::write1());
168 else if (op == "LEFT") program.push_back(PostInstruction::left());
169 else if (op == "RIGHT") program.push_back(PostInstruction::right());
170 else if (op == "HALT") program.push_back(PostInstruction::halt());
171 else if (op == "JZ" || op == "J1") {
172 std::size_t target;
173 if (!(in >> target)) { in.setstate(std::ios::failbit); return in; }
174 program.push_back(op == "JZ" ? PostInstruction::jumpIfZero(target)
175 : PostInstruction::jumpIfOne(target));
176 } else {
177 in.setstate(std::ios::failbit); return in;
178 }
179 }
180 try {
181 PostMachine temp(std::move(program));
182 temp.head_ = head; temp.state_ = state; temp.halted_ = halted != 0;
183 temp.tape_ = std::move(tape);
184 if (state > temp.program_.size()) throw std::invalid_argument("Invalid state.");
185 m = std::move(temp);
186 } catch (...) {
187 in.setstate(std::ios::failbit);
188 }
189 return in;
190}
